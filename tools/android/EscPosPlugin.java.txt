package __APPID__;

import android.annotation.SuppressLint;
import android.app.PendingIntent;
import android.bluetooth.BluetoothAdapter;
import android.bluetooth.BluetoothDevice;
import android.bluetooth.BluetoothManager;
import android.bluetooth.BluetoothSocket;
import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.content.IntentFilter;
import android.hardware.usb.UsbConstants;
import android.hardware.usb.UsbDevice;
import android.hardware.usb.UsbDeviceConnection;
import android.hardware.usb.UsbEndpoint;
import android.hardware.usb.UsbInterface;
import android.hardware.usb.UsbManager;
import android.os.Build;
import android.util.Base64;

import com.getcapacitor.JSArray;
import com.getcapacitor.JSObject;
import com.getcapacitor.PermissionState;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;
import com.getcapacitor.annotation.Permission;
import com.getcapacitor.annotation.PermissionCallback;

import java.io.OutputStream;
import java.net.InetSocketAddress;
import java.net.Socket;
import java.util.Arrays;
import java.util.Set;
import java.util.UUID;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;

// Impresión directa (sin diálogo) de tickets ESC/POS: Bluetooth (SPP), USB y red (puerto 9100).
// La app web arma los bytes del ticket y este plugin solo los envía a la impresora elegida.
@CapacitorPlugin(
    name = "EscPos",
    permissions = { @Permission(alias = "bluetooth", strings = { "android.permission.BLUETOOTH_CONNECT" }) }
)
public class EscPosPlugin extends Plugin {
    private static final UUID SPP = UUID.fromString("00001101-0000-1000-8000-00805F9B34FB");
    private static final String ACCION_USB = "__APPID__.USB_PERMISO";

    private boolean necesitaPermisoBt() {
        return Build.VERSION.SDK_INT >= 31 && getPermissionState("bluetooth") != PermissionState.GRANTED;
    }

    // ---------- Listar dispositivos para elegir en Configuración ----------
    @PluginMethod
    public void listarBluetooth(PluginCall call) {
        if (necesitaPermisoBt()) { requestPermissionForAlias("bluetooth", call, "permisoBtLista"); return; }
        devolverBluetooth(call);
    }

    @PermissionCallback
    private void permisoBtLista(PluginCall call) {
        if (necesitaPermisoBt()) { call.reject("Permiso de Bluetooth denegado. Actívalo en Ajustes > Aplicaciones > permisos."); return; }
        devolverBluetooth(call);
    }

    @SuppressLint("MissingPermission")
    private void devolverBluetooth(PluginCall call) {
        try {
            BluetoothManager bm = (BluetoothManager) getContext().getSystemService(Context.BLUETOOTH_SERVICE);
            BluetoothAdapter ad = bm == null ? null : bm.getAdapter();
            if (ad == null) { call.reject("Este equipo no tiene Bluetooth"); return; }
            if (!ad.isEnabled()) { call.reject("El Bluetooth está apagado. Enciéndelo y vincula la impresora en Ajustes."); return; }
            JSArray arr = new JSArray();
            Set<BluetoothDevice> vinculados = ad.getBondedDevices();
            if (vinculados != null) {
                for (BluetoothDevice d : vinculados) {
                    JSObject o = new JSObject();
                    o.put("nombre", d.getName() == null ? d.getAddress() : d.getName());
                    o.put("direccion", d.getAddress());
                    arr.put(o);
                }
            }
            JSObject r = new JSObject();
            r.put("dispositivos", arr);
            call.resolve(r);
        } catch (Exception e) {
            call.reject(String.valueOf(e.getMessage()));
        }
    }

    @PluginMethod
    public void listarUsb(PluginCall call) {
        try {
            UsbManager um = (UsbManager) getContext().getSystemService(Context.USB_SERVICE);
            JSArray arr = new JSArray();
            if (um != null) {
                for (UsbDevice d : um.getDeviceList().values()) {
                    boolean impresora = false;
                    for (int i = 0; i < d.getInterfaceCount(); i++) {
                        if (d.getInterface(i).getInterfaceClass() == UsbConstants.USB_CLASS_PRINTER) impresora = true;
                    }
                    JSObject o = new JSObject();
                    String nombre = d.getProductName();
                    o.put("nombre", nombre == null ? "Dispositivo USB" : nombre);
                    o.put("direccion", d.getVendorId() + ":" + d.getProductId());
                    o.put("impresora", impresora);
                    arr.put(o);
                }
            }
            JSObject r = new JSObject();
            r.put("dispositivos", arr);
            call.resolve(r);
        } catch (Exception e) {
            call.reject(String.valueOf(e.getMessage()));
        }
    }

    // ---------- Imprimir ----------
    @PluginMethod
    public void imprimir(PluginCall call) {
        String tipo = call.getString("tipo", "");
        if ("bluetooth".equals(tipo) && necesitaPermisoBt()) { requestPermissionForAlias("bluetooth", call, "permisoBtImprimir"); return; }
        ejecutar(call);
    }

    @PermissionCallback
    private void permisoBtImprimir(PluginCall call) {
        if (necesitaPermisoBt()) { call.reject("Permiso de Bluetooth denegado. Actívalo en Ajustes > Aplicaciones > permisos."); return; }
        ejecutar(call);
    }

    private void ejecutar(PluginCall call) {
        try {
            String tipo = call.getString("tipo", "");
            String dir = call.getString("direccion", "");
            String b64 = call.getString("datos", "");
            byte[] datos = Base64.decode(b64, Base64.DEFAULT);
            if (datos.length == 0) { call.reject("No hay nada que imprimir"); return; }
            if ("bluetooth".equals(tipo)) enviarBluetooth(dir, datos);
            else if ("usb".equals(tipo)) enviarUsb(dir, datos);
            else if ("red".equals(tipo)) enviarRed(dir, datos);
            else { call.reject("Tipo de conexión desconocido: " + tipo); return; }
            JSObject r = new JSObject();
            r.put("ok", true);
            call.resolve(r);
        } catch (Exception e) {
            String m = e.getMessage();
            call.reject(m == null || m.length() == 0 ? e.getClass().getSimpleName() : m);
        }
    }

    @SuppressLint("MissingPermission")
    private void enviarBluetooth(String mac, byte[] datos) throws Exception {
        BluetoothManager bm = (BluetoothManager) getContext().getSystemService(Context.BLUETOOTH_SERVICE);
        BluetoothAdapter ad = bm == null ? null : bm.getAdapter();
        if (ad == null || !ad.isEnabled()) throw new Exception("El Bluetooth está apagado");
        BluetoothDevice dev = ad.getRemoteDevice(mac);
        // Cancelar la búsqueda exige el permiso BLUETOOTH_SCAN en Android 12+; no lo necesitamos (solo usamos impresoras ya vinculadas)
        try { ad.cancelDiscovery(); } catch (SecurityException ignorar) { }
        BluetoothSocket s = null;
        try {
            s = dev.createRfcommSocketToServiceRecord(SPP);
            s.connect();
        } catch (Exception primero) {
            try { if (s != null) s.close(); } catch (Exception ignorar) { }
            try {
                s = dev.createInsecureRfcommSocketToServiceRecord(SPP);
                s.connect();
            } catch (Exception segundo) {
                try { if (s != null) s.close(); } catch (Exception ignorar) { }
                throw new Exception("No se pudo conectar con la impresora Bluetooth. ¿Está encendida y cerca?");
            }
        }
        try {
            OutputStream os = s.getOutputStream();
            // Las impresoras Bluetooth económicas tienen poca memoria: se envía en trozos pequeños
            int paso = 480;
            for (int i = 0; i < datos.length; i += paso) {
                os.write(datos, i, Math.min(paso, datos.length - i));
                os.flush();
                Thread.sleep(25);
            }
            Thread.sleep(Math.max(400, datos.length / 6));   // deja terminar de imprimir antes de cerrar
        } finally {
            try { s.close(); } catch (Exception ignorar) { }
        }
    }

    private void enviarRed(String dir, byte[] datos) throws Exception {
        String host = dir.trim();
        int puerto = 9100;
        int i = host.lastIndexOf(':');
        if (i > 0) { puerto = Integer.parseInt(host.substring(i + 1).trim()); host = host.substring(0, i).trim(); }
        Socket s = new Socket();
        try {
            s.connect(new InetSocketAddress(host, puerto), 4000);
            OutputStream os = s.getOutputStream();
            os.write(datos);
            os.flush();
            Thread.sleep(300);
        } catch (Exception e) {
            throw new Exception("No se pudo conectar con la impresora de red " + host + ":" + puerto);
        } finally {
            try { s.close(); } catch (Exception ignorar) { }
        }
    }

    private void pedirPermisoUsb(UsbManager um, UsbDevice dev) throws Exception {
        final CountDownLatch espera = new CountDownLatch(1);
        BroadcastReceiver rx = new BroadcastReceiver() {
            @Override
            public void onReceive(Context c, Intent i) { espera.countDown(); }
        };
        Context ctx = getContext();
        IntentFilter filtro = new IntentFilter(ACCION_USB);
        if (Build.VERSION.SDK_INT >= 33) ctx.registerReceiver(rx, filtro, Context.RECEIVER_NOT_EXPORTED);
        else ctx.registerReceiver(rx, filtro);
        try {
            Intent in = new Intent(ACCION_USB);
            in.setPackage(ctx.getPackageName());
            int flags = Build.VERSION.SDK_INT >= 31 ? PendingIntent.FLAG_MUTABLE : 0;
            PendingIntent pi = PendingIntent.getBroadcast(ctx, 0, in, flags);
            um.requestPermission(dev, pi);
            espera.await(30, TimeUnit.SECONDS);
        } finally {
            try { ctx.unregisterReceiver(rx); } catch (Exception ignorar) { }
        }
        if (!um.hasPermission(dev)) throw new Exception("Permiso USB denegado. Acepta el aviso de Android (marca «usar siempre»).");
    }

    private void enviarUsb(String dir, byte[] datos) throws Exception {
        UsbManager um = (UsbManager) getContext().getSystemService(Context.USB_SERVICE);
        String[] p = dir.split(":");
        if (um == null || p.length != 2) throw new Exception("Impresora USB mal configurada");
        int vid = Integer.parseInt(p[0].trim());
        int pid = Integer.parseInt(p[1].trim());
        UsbDevice dev = null;
        for (UsbDevice d : um.getDeviceList().values()) {
            if (d.getVendorId() == vid && d.getProductId() == pid) { dev = d; break; }
        }
        if (dev == null) throw new Exception("No se encuentra la impresora USB. Revisa el cable y el adaptador OTG.");
        if (!um.hasPermission(dev)) pedirPermisoUsb(um, dev);
        UsbInterface iface = null;
        UsbEndpoint salida = null;
        for (int i = 0; i < dev.getInterfaceCount(); i++) {
            UsbInterface ui = dev.getInterface(i);
            for (int j = 0; j < ui.getEndpointCount(); j++) {
                UsbEndpoint ep = ui.getEndpoint(j);
                if (ep.getType() == UsbConstants.USB_ENDPOINT_XFER_BULK && ep.getDirection() == UsbConstants.USB_DIR_OUT) {
                    if (iface == null || ui.getInterfaceClass() == UsbConstants.USB_CLASS_PRINTER) { iface = ui; salida = ep; }
                }
            }
        }
        if (iface == null || salida == null) throw new Exception("El dispositivo USB no parece una impresora");
        UsbDeviceConnection con = um.openDevice(dev);
        if (con == null) throw new Exception("No se pudo abrir la impresora USB");
        try {
            con.claimInterface(iface, true);
            int paso = 4096;
            for (int i = 0; i < datos.length; i += paso) {
                int n = Math.min(paso, datos.length - i);
                byte[] trozo = Arrays.copyOfRange(datos, i, i + n);
                int r = con.bulkTransfer(salida, trozo, n, 8000);
                if (r < 0) throw new Exception("Error al enviar datos a la impresora USB");
            }
            Thread.sleep(300);
        } finally {
            try { con.releaseInterface(iface); } catch (Exception ignorar) { }
            con.close();
        }
    }
}
